Function that reads one participant’s HR file:

read_hr = function(file, subject, exercise) {
  #read file without assuming column names
  raw = read_csv(file, col_names = FALSE, show_col_types = FALSE)
  #first row = start time
  start_time = ymd_hms(raw[[1]][1], tz = "UTC")
  #second row = sampling frequency
  sample_rate = as.numeric(raw[[1]][2])
  #remaining rows = HR measurements
  hr = as.numeric(raw[[1]][3:nrow(raw)])
  #construct dataset
  tibble(
    subject = subject,
    exercise = exercise,
    session_time = seq(0, by = 1 / sample_rate, length.out = length(hr)),
    timestamp = start_time + session_time,
    HR = hr
  )
}

R can automatically process every folder with this function:

read_exercise_folder = function(folder, exercise_name) {
  subject_folders = list.dirs(folder, recursive = FALSE, full.names = TRUE)
  map_dfr(subject_folders, function(subject_folder) {
    subject = basename(subject_folder)
    hr_file = file.path(subject_folder, "HR.csv")
    #skip folder if HR.csv doesn't exist
    if (!file.exists(hr_file)) {
      return(NULL)
    }
    read_hr(file = hr_file, subject = subject, exercise = exercise_name
    )
  })
}

Now we will create a separate dataset for tags.csv, our timestamps dataset:

read_tags = function(file, subject, exercise) {
  raw = read_csv(file, col_names = FALSE, show_col_types = FALSE)
  tibble(
    subject = subject,
    exercise = exercise,
    tag_number = seq_len(nrow(raw)),
    tag_time = ymd_hms(
      raw[[1]],
      tz = "UTC"
    )
  )
}

Now assign a stage to every row of hr_clean based on which two tag times it falls between. We go session-by-session using findInterval():

hr_labeled = hr_clean %>%
  group_by(subject, protocol, exercise) %>%
  group_modify(~{
    current_tags = tags_labeled %>%
      filter(
        subject == .y$subject,
        protocol == .y$protocol,
        exercise == .y$exercise
      ) %>%
      arrange(tag_time)
    #find which tag interval each HR timestamp belongs to
    interval_id = findInterval(
      .x$timestamp,
      current_tags$tag_time
    )
    #stage before first tag depends on protocol
    pre_stage = case_when(
      .y$protocol == "V1" & .y$exercise == "Aerobic"   ~ "warmup",
      .y$protocol == "V1" & .y$exercise == "Anaerobic" ~ "warmup",
      .y$protocol == "V2"                               ~ "baseline",
      TRUE ~ NA_character_
    )
    stage = ifelse(
      interval_id == 0,
      pre_stage,
      current_tags$stage[pmin(interval_id, nrow(current_tags))]
    )
    .x %>%
      mutate(stage = stage)
  }) %>%
  ungroup()

At this point the analytical dataset is in good shape. We will now create a few recovery summary variables that will make the later comparisons interpretable.

For example, create the HR value at the start of each recovery period:

recovery_data = hr_analysis %>%
  filter(stage_type == "recovery") %>%
  group_by(subject, protocol, exercise, stage, bout_number) %>%
  arrange(recovery_time, .by_group = TRUE) %>%
  mutate(
    recovery_start_HR = first(HR),
    HR_drop = recovery_start_HR - HR
  ) %>%
  ungroup()

This way, HR_drop tells us how many bpm the participant has recovered relative to the beginning of that recovery period:

\(HR_{drop} (t) = HR(0) - HR(t)\)

So if someone starts recovery at 155 bpm and is at 130 bpm after 60 seconds

\(HR_{drop} (60) = 155 - 130 = 25bpm\)

Find largest common window:

windows <- c(60, 90, 120, 150, 180)

coverage_by_window <- purrr::map_dfr(windows, function(w) {
  
  recovery_data %>%
    filter(recovery_time >= 0,
           recovery_time <= w) %>%
    group_by(subject, protocol, exercise, bout_number) %>%
    summarise(
      min_time = min(recovery_time),
      max_time = max(recovery_time),
      n_unique_times = n_distinct(recovery_time),
      .groups = "drop"
    ) %>%
    mutate(
      window = w,
      complete = min_time == 0 &
        max_time >= w &
        n_unique_times >= (w + 1)
    )
  
}) %>%
  group_by(exercise, protocol, window) %>%
  summarise(
    total = n(),
    complete = sum(complete),
    percent_complete = 100 * complete / total,
    .groups = "drop"
  )
